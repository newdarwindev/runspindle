\begin{frame}[label=economics]{Cost of accepted work}
\[
C=M+T+H+V+K+R+F,\qquad U=C/A\quad\text{when }A>0
\]
{\small
\begin{tabular}{p{.14\linewidth}p{.77\linewidth}}
$M,T$ & Model and tool costs \\
$H,V$ & Human task effort and independent verification \\
$K,R$ & Maintenance/support and recovery/repair \\
$F$ & Fixed cost allocated to the period, including setup \\
$A$ & Independently accepted, authorized completions in that period \\
\end{tabular}}
\par\medskip
Each cost appears once, in the same currency and accounting period.
Failures still cost money. At $A=0$, unit cost is undefined, including when $C=0$.
\vfill
{\footnotesize Pending effects, unknown outcomes and correct stops are reported
separately from $A$. Disclose labor rates, allocation and actual spend separately
from elapsed time. Accounting identity, not a price, measurement or forecast.}
\end{frame}
